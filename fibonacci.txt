    //al n-lea nr din fibo 0, 1, 1, 2, 3, 5, 8, 13...
    int fibonacci(int x) {
        if (x <= 1) {
            return x;
        }
        
        int f1;
        int f2;
        f1 = fibonacci(x - 1);
        f2 = fibonacci(x - 2);
        
        return f1 + f2;
    }
