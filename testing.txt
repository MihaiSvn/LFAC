class MathToolbox
{
    class_var_section
        int sir[5];
        int n = 5;

    class_methods_section
        int initSir(int v1, int v2, int v3, int v4, int v5) {
            sir[0] = v1;
            sir[1] = v2;
            sir[2] = v3;
            sir[3] = v4;
            sir[4] = v5;
            return 1;
        }

        int calculeazaSuma() {
            int i;
            int suma = 0;
            for (i = 0; i < n; i++) {
                suma = suma + sir[i];
            }
            return suma;
        }

        bool estePar(int index) {
            int valoare = sir[index];
            if (valoare % 2 == 0) {
                return true;
            }
            return false;
        }
}

global_var_section
    int suma_finala;
    int fact_suma;

global_fun_section
    int factorial(int n) {
        if (n == 0) {
            return 1;
        }
        
        // n * factorial(n - 1)
        auto rez_mic = factorial(n - 1);
        return n * rez_mic;
    }
