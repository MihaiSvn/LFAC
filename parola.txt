class_section

class SecurityBot
{
    class_var_section
        auto pinCorect = 1234;
        int incercariEsuate = 0;
    class_methods_section
        int verificaPin(int cod) {
            if (incercariEsuate >= 3) {
                Print("SISTEM BLOCAT! Prea multe incercari.");
                return 0;
            }

            if (cod == pinCorect) {
                Print("ACCES PERMIS! Bine ai venit.");
                incercariEsuate = 0;
                return 1;
            } else {
                incercariEsuate = incercariEsuate + 1;
                Print("PIN GRESIT!");
                return 0;
            }
        }
}

main
    SecurityBot robot = new SecurityBot();
    
    Print("Incercarea 1 (Gresit):");
    robot.verificaPin(1234);
    Print(newline);
    
    Print("Incercarea 2 (Gresit):");
    robot.verificaPin(2222);
    Print(newline);
    
    Print("Incercarea 3 (Corect):");
    robot.verificaPin(1234);
    Print(newline);
    
    Print("Incercarea 4 (Dupa reset):");
    robot.verificaPin(1234);
    Print(newline);
end_main
