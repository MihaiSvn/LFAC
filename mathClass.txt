class_section

class MathToolbox
{
    class_var_section
        int sir[5];
        int n = 5;

    class_methods_section
        int initSir(int v1, int v2, int v3, int v4, int v5) {
            sir[0] = v1;
            sir[1] = v2;
            sir[2] = v3;
            sir[3] = v4;
            sir[4] = v5;
            return 1;
        }

        int calculeazaSuma() {
            int i;
            int suma = 0;
            for (i = 0; i < n; i++) {
                suma = suma + sir[i];
            }
            return suma;
        }

        bool estePar(int index) {
            int valoare = sir[index];
            if (valoare % 2 == 0) {
                return true;
            }
            return false;
        }
        int gasesteMax() {
            int max;
            int i;
            max = sir[0]; 
            for (i = 1; i < n; i++) {
                if (sir[i] > max) {
                    max = sir[i];
                }
            }
            return max;
        }
        int inverseazaSir() {
            int i;
            int temp;
            int limita;
            limita = n / 2; 

            for (i = 0; i < limita; i++) {
                temp = sir[i];
                sir[i] = sir[n - 1 - i];
                sir[n - 1 - i] = temp;
            }
            return 1;
        }
        int mediaSir() {
            int suma;
            int i;
            suma = 0;
            for (i = 0; i < n; i++) {
                suma = suma + sir[i];
            }
            return suma / n;
        }
        int printMethodVector(int lungime) {
        int i;
        for(i = 0; i < lungime; i++) {
            Print(sir[i]);
        }
        Print(newline);
        return 1;
    }
}

global_var_section
    int suma_finala;
    int fact_suma;
    int m,medie;
    int sir[10] = {56, 2, 38, 12, 91, 8, 72, 5, 23, 16};

global_fun_section
    int factorial(int n) {
        if (n == 0) {
            return 1;
        }
        
        // n * factorial(n - 1)
        auto rez_mic = factorial(n - 1);
        return n * rez_mic;
    }
    int bubbleSort(int lungime) {
        int i;
        int j;
        int temp;
        
        for (i = 0; i < lungime - 1; i++) {
            for (j = 0; j < lungime - i - 1; j++) {
                if (sir[j] > sir[j + 1]) {
                    temp = sir[j];
                    sir[j] = sir[j + 1];
                    sir[j + 1] = temp;
                }
            }
        }
        return 1;
    }

    int printVector(int lungime) {
        int i;
        for(i = 0; i < lungime; i++) {
            Print(sir[i]);
        }
        Print(newline);
        return 1;
    }


main
    MathToolbox tool = new MathToolbox();
    Print("Initializam vectorul din clasa...");
    tool.initSir(10, 50, 20, 40, 30);
    tool.printMethodVector(5);

    suma_finala = tool.calculeazaSuma();
    Print("Suma elementelor din clasa este:");
    Print(suma_finala); // 15
    Print(newline);

    if (tool.estePar(1)) {
        Print("Elementul de la index 1 este PAR");
    } else {
        Print("Elementul de la index 1 este IMPAR");
    }
    Print(newline);

    m = tool.gasesteMax();

    medie = tool.mediaSir();
