    int cmmdc(int a, int b) {
        if (a == 0) {
            return b;
        }
        
        while (b != 0) {
            if (a > b) {
                a = a - b;
            } else {
                b = b - a;
            }
        }
        
        return a;
    }

    int bubbleSort(int lungime) {
        int i;
        int j;
        int temp;
        
        for (i = 0; i < lungime - 1; i++) {
            for (j = 0; j < lungime - i - 1; j++) {
                // Dacă elementul curent e mai mare decât următorul, le inversăm
                if (sir[j] > sir[j + 1]) {
                    temp = sir[j];
                    sir[j] = sir[j + 1];
                    sir[j + 1] = temp;
                }
            }
        }
        return 1;
    }
